const GROQ_KEY = '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';
const NPC_PERSONALITY = 'Eres Lyra, guardiana del primer mundo de Aura. Eres curiosa, amigable y misteriosa. Sabes sobre el metaverso de Aura, sus mundos y su economía. Responde en español, máximo 2 oraciones.';

const scene = new THREE.Scene();
scene.background = new THREE.Color(0x0a0a1a);
scene.fog = new THREE.Fog(0x0a0a1a, 20, 60);

const camera = new THREE.PerspectiveCamera(75, innerWidth/innerHeight, 0.1, 100);
camera.position.set(0, 4, 8);

const renderer = new THREE.WebGLRenderer({ canvas: document.getElementById('c'), antialias: true });
renderer.setSize(innerWidth, innerHeight);
renderer.shadowMap.enabled = true;

const ambient = new THREE.AmbientLight(0x334466, 0.8);
scene.add(ambient);
const dirLight = new THREE.DirectionalLight(0x8877ff, 1.2);
dirLight.position.set(5, 10, 5);
dirLight.castShadow = true;
scene.add(dirLight);
const pointLight = new THREE.PointLight(0x00ffaa, 1.5, 15);
pointLight.position.set(0, 3, 0);
scene.add(pointLight);

const floorGeo = new THREE.PlaneGeometry(40, 40, 20, 20);
const floorMat = new THREE.MeshLambertMaterial({ color: 0x111133 });
const floor = new THREE.Mesh(floorGeo, floorMat);
floor.rotation.x = -Math.PI / 2;
floor.receiveShadow = true;
scene.add(floor);

const gridHelper = new THREE.GridHelper(40, 40, 0x222255, 0x222244);
scene.add(gridHelper);

function makeBox(w, h, d, color, x, y, z) {
  const m = new THREE.Mesh(
    new THREE.BoxGeometry(w, h, d),
    new THREE.MeshLambertMaterial({ color })
  );
  m.position.set(x, y, z);
  m.castShadow = true;
  m.receiveShadow = true;
  scene.add(m);
  return m;
}

makeBox(20, 0.3, 0.3, 0x534AB7, 0, 0.15, -10);

const playerGeo = new THREE.CylinderGeometry(0.3, 0.3, 1.6, 8);
const playerMat = new THREE.MeshLambertMaterial({ color: 0x7F77DD });

const keys = {};
document.addEventListener('keydown', e => keys[e.key.toLowerCase()] = true);
document.addEventListener('keyup', e => keys[e.key.toLowerCase()] = false);

const chatBox = document.getElementById('chat-box');
const messagesDiv = document.getElementById('messages');
const chatInput = document.getElementById('chat-input');
const npcLabel = document.getElementById('npc-label');
let chatOpen = false;
let npcNear = false;
const chatHistory = [];

function addMsg(text, type) {
  const d = document.createElement('div');
  d.className = 'msg ' + type;
  d.textContent = (type === 'npc' ? 'Lyra: ' : type === 'player' ? 'Tú: ' : '') + text;
  messagesDiv.appendChild(d);
  messagesDiv.scrollTop = messagesDiv.scrollHeight;
}

addMsg('Acércate al NPC verde para hablar con Lyra.', 'system');

async function askNPC(userMsg) {
  chatHistory.push({ role: 'user', content: userMsg });
  addMsg('...', 'npc');
  try {
    const res = await fetch('https://api.groq.com/openai/v1/chat/completions', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Authorization': 'Bearer ' + GROQ_KEY
      },
      body: JSON.stringify({
        model: 'llama3-8b-8192',
        messages: [{ role: 'system', content: NPC_PERSONALITY }, ...chatHistory],
        max_tokens: 120
      })
    });
    const data = await res.json();
    const reply = data.choices?.[0]?.message?.content || 'Lyra está pensando...';
    chatHistory.push({ role: 'assistant', content: reply });
    messagesDiv.lastChild.remove();
    addMsg(reply, 'npc');
  } catch(e) {
    messagesDiv.lastChild.remove();
    addMsg('(Error de conexión con la IA)', 'system');
  }
}

document.addEventListener('keydown', e => {
  if (e.key === 'Enter' && npcNear && !chatOpen) {
    chatOpen = true;
    chatBox.style.display = 'block';
    chatInput.focus();
    if (chatHistory.length === 0) askNPC('Hola, ¿quién eres?');
  }
  if (e.key === 'Escape' && chatOpen) {
    chatOpen = false;
    chatBox.style.display = 'none';
  }
});

chatInput.addEventListener('keydown', async e => {
  if (e.key === 'Enter' && chatInput.value.trim()) {
    const msg = chatInput.value.trim();
    chatInput.value = '';
    addMsg(msg, 'player');
    await askNPC(msg);
  }
});

const clock = new THREE.Clock();
let t = 0;

function animate() {
  requestAnimationFrame(animate);
  const delta = clock.getDelta();
  t += delta;

  if (!chatOpen) {
    const speed = 5;
    const dir = new THREE.Vector3();
    if (keys['w'] || keys['arrowup']) dir.z -= 1;
    if (keys['s'] || keys['arrowdown']) dir.z += 1;
    if (keys['a'] || keys['arrowleft']) dir.x -= 1;
    if (keys['d'] || keys['arrowright']) dir.x += 1;
    if (dir.length() > 0) {
      dir.normalize().multiplyScalar(speed * delta);
      player.position.add(dir);
      player.position.x = Math.max(-9, Math.min(9, player.position.x));
      player.position.z = Math.max(-9, Math.min(9, player.position.z));
      player.rotation.y = Math.atan2(dir.x, dir.z);
    }
  }

  npcGroup.rotation.y = t * 0.5;
  npcLight.intensity = 0.8 + Math.sin(t * 2) * 0.4;

  const dist = player.position.distanceTo(npcGroup.position);
  npcNear = dist < 3;

  if (npcNear && !chatOpen) {
    npcLabel.style.display = 'block';
    const pos = npcGroup.position.clone();
    pos.y += 2.5;
    pos.project(camera);
    npcLabel.style.left = ((pos.x + 1) / 2 * innerWidth - 60) + 'px';
    npcLabel.style.top = ((-pos.y + 1) / 2 * innerHeight) + 'px';
  } else {
    npcLabel.style.display = 'none';
  }

  camera.position.x += (player.position.x - camera.position.x) * 0.08;
  camera.position.z += (player.position.z + 8 - camera.position.z) * 0.08;
  camera.lookAt(player.position.x, 1, player.position.z);

  renderer.render(scene, camera);
}

window.addEventListener('resize', () => {
  camera.aspect = innerWidth / innerHeight;
  camera.updateProjectionMatrix();
  renderer.setSize(innerWidth, innerHeight);
});

document.getElementById('loading').style.display = 'none';
animate();
